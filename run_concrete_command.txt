# =========================================================================
# CONCRETE2YOU - AUTOMATION RUN COMMANDS LIST
# =========================================================================

# 1. LOGIN
# Run headed browser login automation (ends on My Account page):
npm run open

# Run headless login test suite:
npm run test


# 2. CREATE ACCOUNT
# Run headed browser account creation automation (registers unique 2-digit serial emails):
npm run create-account

# Run headless account creation test suite:
npm run test-register


# 3. LOGIN WITH CHECKOUT
# Run headed browser Login with Checkout automation (logs in, adds mix, inputs postcode, resolves dates, ends on checkout):
npm run checkout

# Run headless Login with Checkout test suite:
npm run test-checkout


# 4. CONCRETE PRODUCT SELECTION & BASKET ADDITION
# Run headed browser concrete product selection automation (navigates menus, checks postcode, configures dynamic inputs, adds to basket):
npm run add-concrete

# Run headless concrete product selection test suite:
npm run test-add-concrete


# 5. WITHOUT LOGIN CHECKOUT
# Run headed browser Without Login Checkout (Guest Checkout) automation (adds product, proceeds as guest, enters shipping details, ends on payment):
npm run checkout-guest

# Run headless Without Login Checkout test suite:
npm run test-checkout-guest


# =========================================================================
# HELPER UTILITIES
# =========================================================================

# Save login session cookies manually (useful for manual CAPTCHA bypass):
npm run save-auth

# Pull latest automation code updates from GitHub:
git pull

# Commit and push your local code changes to GitHub:
git add . ; git commit -m "my updates" ; git push
